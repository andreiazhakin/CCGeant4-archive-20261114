cmake_minimum_required(VERSION 3.20)
project(CCDiracSubsetExample LANGUAGES CXX)
find_package(Geant4 11.4 REQUIRED)
include(${Geant4_USE_FILE})
add_executable(dirac_subset_example transport.cc)
target_compile_features(dirac_subset_example PRIVATE cxx_std_17)
target_compile_definitions(dirac_subset_example PRIVATE NOMINMAX)
target_link_libraries(dirac_subset_example PRIVATE ${Geant4_LIBRARIES})
if(WIN32)
  target_link_libraries(dirac_subset_example PRIVATE Psapi)
endif()
