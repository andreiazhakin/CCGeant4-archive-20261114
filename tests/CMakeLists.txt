cmake_minimum_required(VERSION 3.20)
project(CCDiracValidation LANGUAGES CXX)
find_package(Geant4 11.4 REQUIRED)
include(${Geant4_USE_FILE})
add_executable(dirac_transport dirac_transport.cc)
target_compile_features(dirac_transport PRIVATE cxx_std_17)
target_compile_definitions(dirac_transport PRIVATE NOMINMAX)
target_link_libraries(dirac_transport PRIVATE ${Geant4_LIBRARIES})
if(WIN32)
  target_link_libraries(dirac_transport PRIVATE Psapi)
endif()
install(TARGETS dirac_transport RUNTIME DESTINATION bin)
foreach(test_name audit_isotopes audit_global_domain table_kernel_probe)
  add_executable(${test_name} ${test_name}.cc)
  target_compile_features(${test_name} PRIVATE cxx_std_17)
  target_compile_definitions(${test_name} PRIVATE NOMINMAX)
  target_link_libraries(${test_name} PRIVATE ${Geant4_LIBRARIES})
  if(WIN32)
    target_link_libraries(${test_name} PRIVATE Psapi)
  endif()
  install(TARGETS ${test_name} RUNTIME DESTINATION bin)
endforeach()
add_subdirectory(audit_trackless)
install(TARGETS audit_trackless RUNTIME DESTINATION bin)
add_subdirectory(audit_rangecut)
install(TARGETS audit_rangecut RUNTIME DESTINATION bin)
