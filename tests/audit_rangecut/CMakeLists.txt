cmake_minimum_required(VERSION 3.20)
project(CCAuditRangeCut LANGUAGES CXX)
find_package(Geant4 11.4 REQUIRED)
include(${Geant4_USE_FILE})
add_executable(audit_rangecut audit_rangecut.cc)
target_compile_features(audit_rangecut PRIVATE cxx_std_17)
target_compile_definitions(audit_rangecut PRIVATE NOMINMAX)
target_link_libraries(audit_rangecut PRIVATE ${Geant4_LIBRARIES})
